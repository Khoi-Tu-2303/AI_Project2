\chapter{Các thuật toán suy luận}
\label{chap:cac-thuat-toan}

\section{Thuật toán DPLL giải bài toán SAT}
\label{sec:thuat-toan-dpll}

Mọi kết luận của hệ thống đều quy về một câu hỏi duy nhất: một công thức dạng chuẩn hội (CNF) có thỏa được hay không. Mục này trình bày bộ giải SAT được cài đặt thủ công theo thủ tục Davis--Putnam--Logemann--Loveland (DPLL), gồm ba thành phần: lan truyền đơn vị, chọn biến để rẽ nhánh và quay lui khi gặp mâu thuẫn.

\subsection{Bài toán SAT và cách biểu diễn}
\label{subsec:bai-toan-sat}

Cho một công thức CNF $F = K_1 \land K_2 \land \dots \land K_m$ trên tập $N$ biến mệnh đề $\{x_1, \dots, x_N\}$, trong đó mỗi mệnh đề $K_j$ là một phép tuyển của các literal (biến hoặc phủ định của biến). Bài toán SAT yêu cầu quyết định tồn tại hay không một phép gán $\alpha : \{x_1,\dots,x_N\} \to \{\text{True}, \text{False}\}$ làm cho mọi mệnh đề đều đúng. Nếu tồn tại, $F$ là \emph{thỏa được} (SAT) và $\alpha$ được gọi là một \emph{mô hình}; ngược lại $F$ là \emph{không thỏa được} (UNSAT).

Trong đồ án, công thức $F$ chính là cơ sở tri thức $KB_t$ đã được mã hóa ở Chương~\ref{chap:hinh-thuc-hoa}, và các biến $x_1, \dots, x_{n^2}$ là các biến cơ sở $C_i$ của từng ô cờ, còn các biến từ $x_{n^2+1}$ trở đi là biến phụ, với $n$ là kích thước bàn cờ. Như vậy tổng số biến là $N = n^2 + n_{\text{phụ}}$, trong đó $n_{\text{phụ}}$ là số biến phụ. Bộ giải sử dụng các quy ước biểu diễn sau:
\begin{itemize}
    \item Literal là số nguyên khác 0 có dấu: $+v$ biểu diễn biến $x_v$ nhận giá trị True (Tội phạm) và $-v$ biểu diễn $x_v$ nhận giá trị False (Vô tội). Mệnh đề là danh sách các literal.
    \item Phép gán hiện thời được lưu trong một mảng $\alpha[v] \in \{0, +1, -1\}$, với $0$ nghĩa là biến chưa được gán.
    \item Mọi literal được gán đều được ghi vào một ngăn xếp gọi là \emph{trail} theo đúng thứ tự gán. Khi quay lui, bộ giải chỉ việc lấy các literal ở đỉnh trail ra và đặt lại $\alpha[v] = 0$, nên không cần sao chép phép gán ở mỗi nhánh.
\end{itemize}

\subsection{Khởi tạo và trạng thái gốc}
\label{subsec:dpll-khoi-tao}

Bộ giải nhận công thức một lần duy nhất khi khởi tạo và chuẩn bị một \emph{trạng thái gốc} dùng chung cho mọi lần giải sau đó. Quá trình khởi tạo gồm các bước:
\begin{enumerate}
    \item \textbf{Chuẩn hóa từng mệnh đề.} Loại các literal lặp lại trong cùng một mệnh đề; loại hẳn các mệnh đề hằng đúng (chứa đồng thời $x$ và $\neg x$); kiểm tra mỗi literal phải là số nguyên khác 0 với $|\ell| \le N$, nếu không báo lỗi.
    \item \textbf{Phân loại mệnh đề.} Mệnh đề rỗng làm cho $F$ không thỏa được ngay lập tức. Mệnh đề đơn vị (một literal) được ghi nhận là \emph{sự thật gốc}. Các mệnh đề còn lại có từ hai literal trở lên được lưu vào kho mệnh đề.
    \item \textbf{Lan truyền gốc.} Gán toàn bộ sự thật gốc rồi chạy lan truyền đơn vị một lần. Độ dài trail sau bước này được ghi nhớ; mọi lần gọi giải về sau đều bắt đầu từ trạng thái này và quay về đúng trạng thái này khi kết thúc.
    \item \textbf{Thứ tự chọn biến.} Các biến do bên gọi chỉ định được đặt lên trước, tiếp theo là các biến còn lại theo thứ tự tăng dần. Bộ suy diễn của đồ án chỉ định các biến cơ sở $x_1,\dots,x_{n^2}$ làm biến quyết định, vì các biến phụ đã được xác định hàm theo biến cơ sở và không cần phải thử cả hai giá trị. Nhờ vậy, không gian tìm kiếm nhánh giảm từ $2^{N}$ xuống tối đa $2^{n^2}$.
\end{enumerate}

Do các sự thật gốc (thân phận của các ô đã được chứng minh) chỉ cần lan truyền một lần, các truy vấn lặp lại trên cùng một cơ sở tri thức không phải trả lại chi phí này.

\subsection{Lan truyền đơn vị}
\label{subsec:lan-truyen-don-vi}

Với mõi phép gán hiện tại, một mệnh đề ở một trong bốn trạng thái: đã được thỏa (có ít nhất một literal đúng), bị vi phạm (mọi literal đều sai), đơn vị (đúng một literal chưa gán, các literal còn lại đều sai) hoặc chưa quyết định. Trong trạng thái đơn vị, literal duy nhất chưa gán buộc phải nhận giá trị True, nếu không mệnh đề sẽ bị vi phạm; việc gán này là hệ quả tất yếu của $F$ và phép gán hiện tại, nên không làm mất nghiệm nào.

\begin{algorithm}[H]
\caption{Lan truyền đơn vị \textsc{UnitPropagate}}
\label{alg:unit-propagate}
\begin{algorithmic}[1]
\REQUIRE Kho mệnh đề $\mathcal{K}$ (độ dài $\ge 2$), phép gán hiện tại $\alpha$, trail $T$
\ENSURE \textbf{True} nếu không có xung đột; \textbf{False} nếu có mệnh đề bị vi phạm
\REPEAT
    \STATE $\textit{changed} \leftarrow \text{False}$
    \FORALL{mệnh đề $K \in \mathcal{K}$}
        \IF{$K$ có một literal đúng dưới $\alpha$}
            \STATE \textbf{continue}
        \ENDIF
        \STATE $U \leftarrow$ tập các literal chưa gán của $K$
        \IF{$U = \emptyset$}
            \RETURN \textbf{False} \COMMENT{mọi literal đều sai: xung đột}
        \ELSIF{$|U| = 1$}
            \STATE gán $\ell \in U$ bằng True; ghi $\ell$ vào $T$
            \STATE $\textit{changed} \leftarrow \text{True}$
        \ENDIF
    \ENDFOR
\UNTIL{$\textit{changed} = \text{False}$}
\RETURN \textbf{True}
\end{algorithmic}
\end{algorithm}

Thủ tục quét lặp lại toàn bộ kho mệnh đề cho đến khi không còn literal nào được gán thêm. Mỗi lượt quét tốn thời gian tuyến tính theo tổng số literal $S$ của công thức, nên một lần lan truyền tốn $O(N \cdot S)$ trong trường hợp xấu nhất.

\subsection{Chọn biến, rẽ nhánh và quay lui}
\label{subsec:dpll-re-nhanh}

Khi lan truyền đơn vị kết thúc mà chưa có xung đột và vẫn còn biến chưa gán, bộ giải chọn biến chưa gán đầu tiên theo thứ tự đã dựng ở Mục~\ref{subsec:dpll-khoi-tao} rồi thử lần lượt hai giá trị. Giá trị False (Vô tội) được thử trước, giá trị True (Tội phạm) được thử sau. Thứ tự cố định này làm cho kết quả, kể cả mô hình được tìm thấy, hoàn toàn tất định và có thể tái lập.

\begin{algorithm}[H]
\caption{Thủ tục đệ quy \textsc{DPLL}}
\label{alg:dpll}
\begin{algorithmic}[1]
\REQUIRE Lan truyền đơn vị đã hoàn tất và không có xung đột
\ENSURE SAT hoặc UNSAT
\STATE $v \leftarrow$ biến chưa gán đầu tiên theo thứ tự chọn biến
\IF{không còn biến chưa gán}
    \RETURN SAT \COMMENT{phép gán đầy đủ, không có xung đột}
\ENDIF
\STATE $\textit{mark} \leftarrow |T|$ \COMMENT{điểm quay lui}
\FORALL{$\ell \in (\neg v,\ v)$}
    \STATE gán $\ell$ bằng True; ghi $\ell$ vào $T$
    \IF{\textsc{UnitPropagate} trả về True}
        \IF{\textsc{DPLL} trả về SAT}
            \RETURN SAT \COMMENT{tìm được mô hình: dừng ngay}
        \ENDIF
    \ENDIF
    \STATE hoàn tác trail $T$ về độ dài $\textit{mark}$ \COMMENT{quay lui}
\ENDFOR
\RETURN UNSAT \COMMENT{cả hai giá trị của $v$ đều thất bại}
\end{algorithmic}
\end{algorithm}

Điểm đáng chú ý của thủ tục là cách quản lý trạng thái. Khi một nhánh thất bại, mọi literal được gán kể từ điểm $\textit{mark}$ (gồm literal quyết định và các literal được lan truyền sau đó) đều bị hoàn tác trong một thao tác duy nhất; nhờ đó khi thủ tục trả về UNSAT, trạng thái bộ giải đúng bằng trạng thái lúc bắt đầu gọi. Ngược lại, khi trả về SAT, phép gán đầy đủ được giữ nguyên để hàm bên ngoài đọc mô hình trước khi dọn dẹp. Vì mỗi lời gọi đệ quy gán thêm ít nhất một biến, độ sâu đệ quy không vượt quá $N$.

\subsection{Giải với giả định và tái sử dụng bộ giải}
\label{subsec:dpll-gia-dinh}

Bài toán suy diễn ở Chương~\ref{chap:hinh-thuc-hoa} đòi hỏi hàng chục đến hàng trăm truy vấn dạng $KB_t \land \ell$ trên cùng một cơ sở tri thức, trong đó $\ell$ là literal của một ô cờ. Thay vì dựng lại bộ giải cho từng truy vấn, giao diện giải nhận một dãy \emph{giả định} (các literal đơn vị tạm thời) và quyết định tính thỏa mãn.

\begin{algorithm}[H]
\caption{Giải với giả định \textsc{Solve}}
\label{alg:solve}
\begin{algorithmic}[1]
\REQUIRE Danh sách giả định $A$
\ENSURE Kết quả $(\textit{status}, \textit{model})$
\IF{$F$ đã không thỏa được ở trạng thái gốc}
    \RETURN UNSAT
\ENDIF
\STATE gán lần lượt các literal trong $A$; nếu có literal đã sai thì xung đột
\IF{không xung đột \AND \textsc{UnitPropagate} trả về True}
    \STATE $\textit{status} \leftarrow \textsc{DPLL}$
    \IF{$\textit{status} = $ SAT}
        \STATE đọc mô hình
    \ENDIF
\ELSE
    \STATE $\textit{status} \leftarrow$ UNSAT
\ENDIF
\STATE hoàn tác trail về trạng thái gốc \COMMENT{luôn thực hiện, kể cả khi lỗi}
\RETURN $(\textit{status}, \textit{model})$
\end{algorithmic}
\end{algorithm}

Bước hoàn tác về trạng thái gốc luôn được thực hiện, kể cả khi xảy ra ngoại lệ, nên một đối tượng bộ giải có thể được gọi lại nhiều lần với các giả định khác nhau mà không có rò rỉ trạng thái giữa các truy vấn.

\subsection{Tính đúng đắn và độ phức tạp}
\label{subsec:dpll-dung-dan}

\textbf{Tính đúng đắn.} Lan truyền đơn vị chỉ gán những literal là hệ quả tất yếu của công thức và phép gán hiện thời, nên không loại bỏ bất kỳ mô hình nào. Thủ tục rẽ nhánh thử cả hai giá trị của mỗi biến quyết định trước khi kết luận UNSAT nên không bỏ sót nhánh nào. Khi mọi biến đã được gán và không có mệnh đề nào bị vi phạm, phép gán đó thỏa mọi mệnh đề, nên kết luận SAT luôn đi kèm một mô hình hợp lệ. Thủ tục luôn dừng vì mỗi lời gọi đệ quy làm tăng thực sự số biến đã gán.

\textbf{Độ phức tạp.} Trong trường hợp xấu nhất, DPLL duyệt $O(2^{n^2})$ nhánh, vì chỉ có $n^2$ biến cơ sở của bàn cờ $n \times n$ được dùng làm biến quyết định (các biến phụ do lan truyền đơn vị suy ra). Mỗi nhánh tốn một lần lan truyền với độ phức tạp $O(N \cdot S)$; đây là hệ quả của việc SAT là bài toán NP-complete và bộ giải không có kỹ thuật học mệnh đề (\emph{clause learning}) hay chọn biến theo heuristic.
%  Tuy nhiên, các gợi ý của trò chơi sinh ra những công thức có cấu trúc rất chặt: sau khi một vài ô được xác định, lan truyền đơn vị thường ép được phần lớn các ô còn lại. Điều này thể hiện rõ ở các số liệu thực nghiệm của Chương~\ref{chap:thuc-nghiem-danh-gia}, khi toàn bộ các hồ sơ vụ án thử nghiệm đều được giải với số lần quay lui bằng 0.

\section{Thuật toán suy diễn logic và phân loại trạng thái}
\label{sec:suy-dien-logic}

Bộ giải DPLL ở Mục~\ref{sec:thuat-toan-dpll} chỉ trả lời câu hỏi "công thức có thỏa được hay không". Nguyên lý hệ quả logic bằng phương pháp phản chứng và ba nhãn phân loại của một ô (Tội phạm, Vô tội, Chưa xác định) đã được phát biểu ở Mục~\ref{sec:bien-menh-de-kb}. Mục này không nhắc lại mà chỉ trình bày cách hiện thực chúng trên bộ giải DPLL và cách giảm số lần gọi bộ giải.

\subsection{Quy về các lần gọi bộ giải}
\label{subsec:he-qua-logic}

Theo nguyên lý phản chứng, $KB \models \ell \iff KB \land \neg\ell$ không thỏa được, với $\ell$ là literal của một biến cơ sở. Mỗi câu hỏi suy diễn do đó là \emph{một} lần gọi bộ giải với đúng một giả định $\neg\ell$: kết quả UNSAT nghĩa là $\ell$ được chứng minh, kết quả SAT nghĩa là chưa chứng minh được và mô hình trả về là một phản ví dụ.

Cần kiểm tra tính thỏa được của $KB$ trước mọi phép phân loại, vì nếu $KB$ không thỏa được thì $KB \land \neg\ell$ cũng không thỏa được với mọi $\ell$, tức là $KB$ ``suy ra mọi thứ'' và kết luận như vậy vô nghĩa. Khi đó hệ thống báo trạng thái \textsc{Inconsistent} và không đưa ra phán quyết nào.

\subsection{Phân loại một ô bằng hai truy vấn}
\label{subsec:phan-loai-mot-o}

Với ô $i$ ứng với biến $v$, thuật toán dưới đây hiện thực trực tiếp ba nhãn nói trên bằng hai truy vấn $KB \land v$ và $KB \land \neg v$, cho biết ô $i$ \emph{có thể} nhận giá trị nào.

\begin{algorithm}[H]
\caption{Phân loại một ô \textsc{ClassifyCell}}
\label{alg:classify-cell}
\begin{algorithmic}[1]
\REQUIRE Cơ sở tri thức $KB$ đã nạp vào bộ giải; ô $i$ ứng với biến cơ sở $v$
\ENSURE \textsc{Criminal}, \textsc{Innocent}, \textsc{Unknown} hoặc \textsc{Inconsistent}
\IF{$\textsc{Solve}([\,])$ không phải SAT}
    \RETURN \textsc{Inconsistent} \COMMENT{$KB$ mâu thuẫn}
\ENDIF
\STATE $c_T \leftarrow (\textsc{Solve}([v])$ là SAT$)$ \COMMENT{ô $i$ có thể là Tội phạm?}
\STATE $c_F \leftarrow (\textsc{Solve}([\neg v])$ là SAT$)$ \COMMENT{ô $i$ có thể là Vô tội?}
\IF{$c_T$ \AND $c_F$}
    \RETURN \textsc{Unknown}
\ELSIF{$c_T$}
    \RETURN \textsc{Criminal}
\ELSE
    \RETURN \textsc{Innocent}
\ENDIF
\end{algorithmic}
\end{algorithm}

Cách làm này đúng nhưng tốn ba lần gọi bộ giải cho mỗi ô (một lần kiểm tra nhất quán và hai truy vấn). Khi phải phân loại tất cả các ô chưa lật ở mỗi bước suy luận, chi phí này có thể giảm đáng kể như trình bày dưới đây.

\subsection{Phân loại đồng thời các ô chưa lật}
\label{subsec:phan-loai-dong-thoi}

Nhận xét then chốt là mô hình mà bộ giải trả về khi SAT là một phép gán \emph{đầy đủ} trên mọi biến $1..N$. Do đó mỗi mô hình tìm được là một \emph{bằng chứng} cho giá trị của \emph{mọi} ô chưa lật cùng lúc: nếu trong một mô hình ô $i$ là Tội phạm thì thân phận Tội phạm của ô $i$ chắc chắn phù hợp với $KB$, và không cần hỏi lại câu $KB \land v$. Hơn nữa, nếu qua các mô hình khác nhau ta đã quan sát được cả hai giá trị của một ô thì ô đó chắc chắn là Chưa xác định mà không cần thêm truy vấn nào. Chỉ khi một ô mới chỉ xuất hiện với một giá trị $w$, ta mới cần hỏi giá trị đối lập $\neg w$.

\begin{algorithm}[H]
\caption{Phân loại đồng thời \textsc{ClassifyCells}}
\label{alg:classify-cells}
\begin{algorithmic}[1]
\REQUIRE Cơ sở tri thức $KB$ đã nạp vào bộ giải; danh sách các ô chưa lật $C$, ô $i \in C$ ứng với biến cơ sở $v_i$
\ENSURE Nhãn của từng ô thuộc $C$, hoặc \textsc{Inconsistent}
\STATE $(s, M) \leftarrow \textsc{Solve}([\,])$
\IF{$s$ không phải SAT}
    \RETURN \textsc{Inconsistent}
\ENDIF
\STATE $V_i \leftarrow \{M[v_i]\}$ với mọi $i \in C$ \COMMENT{tập giá trị đã quan sát của ô $i$}
\FORALL{$i \in C$ theo thứ tự đã cho}
    \IF{$|V_i| = 2$}
        \STATE $\textit{nhãn}(i) \leftarrow$ \textsc{Unknown}
    \ELSE
        \STATE $w \leftarrow$ phần tử duy nhất của $V_i$
        \STATE $(s, M') \leftarrow \textsc{Solve}([\,\text{literal của } v_i \text{ ứng với giá trị } \neg w\,])$
        \IF{$s$ là SAT}
            \STATE $V_j \leftarrow V_j \cup \{M'[v_j]\}$ với mọi $j \in C$ \COMMENT{ghi nhận thêm bằng chứng}
            \STATE $\textit{nhãn}(i) \leftarrow$ \textsc{Unknown}
        \ELSIF{$w = \text{True}$}
            \STATE $\textit{nhãn}(i) \leftarrow$ \textsc{Criminal}
        \ELSE
            \STATE $\textit{nhãn}(i) \leftarrow$ \textsc{Innocent}
        \ENDIF
    \ENDIF
\ENDFOR
\RETURN $\textit{nhãn}$
\end{algorithmic}
\end{algorithm}

\textbf{Tính đúng đắn.} Một ô được gán nhãn \textsc{Unknown} khi đã có hai mô hình của $KB$ gán cho nó hai giá trị khác nhau, nên cả hai thân phận đều thực sự phù hợp với $KB$. Một ô được gán nhãn \textsc{Criminal} (hoặc \textsc{Innocent}) khi truy vấn giá trị đối lập cho kết quả UNSAT, tức là $KB \land \neg \ell$ không thỏa được, nên $KB \models \ell$. Như vậy nhãn của mỗi ô đều được chứng minh, không có nhãn nào dựa trên phỏng đoán.

\textbf{Số lần gọi bộ giải.} Thuật toán gọi bộ giải một lần để lấy mô hình đầu tiên, và sau đó tối đa một lần cho mỗi ô, tức tối đa $1 + |C|$ lần, so với $1 + 2|C|$ lần nếu hỏi hai truy vấn cho từng ô.
\subsection{Từ trạng thái công khai đến báo cáo phân loại}
\label{subsec:tu-trang-thai-den-bao-cao}

Bộ phân loại mà động cơ trò chơi sử dụng nhận vào \emph{chỉ} trạng thái công khai tại một thời điểm (các gợi ý đã lật và các thân phận đã được chứng minh) và thực hiện chuỗi bước sau:
\begin{enumerate}
    \item Mã hóa các gợi ý đã lật và các thân phận đã biết thành $KB_t$ ở dạng CNF theo Chương~\ref{chap:hinh-thuc-hoa}.
    \item Dựng một bộ giải DPLL cho $KB_t$, với các biến cơ sở là biến quyết định.
    \item Phân loại mọi ô chưa lật bằng thuật toán phân loại đồng thời ở trên.
    \item Đóng gói kết quả thành một báo cáo mang theo định danh của đúng trạng thái đã dùng để suy diễn, để động cơ trò chơi từ chối các báo cáo cũ.
\end{enumerate}
Nếu $KB_t$ mâu thuẫn thì báo cáo mang trạng thái \textsc{Inconsistent} và không chứa phân loại nào cho các ô.

Mỗi khi có thêm một ô được lật, hệ thống có thêm một gợi ý và một thân phận mới nên $KB_{t+1}$ khác $KB_t$; vì vậy bộ giải được dựng lại cho mỗi bước suy luận và việc tái sử dụng bộ giải chỉ diễn ra trong phạm vi một lần phân loại.
\section{Thuật toán kiểm tra tính duy nhất của nghiệm}
\label{sec:kiem-tra-tinh-duy-nhat}

Một câu đố hợp lệ phải có \emph{đúng một} nghiệm khi xét toàn bộ tập gợi ý. Nếu không có nghiệm nào, các gợi ý mâu thuẫn với nhau; nếu có nhiều nghiệm, người chơi dù suy luận đúng đến đâu cũng không thể xác định duy nhất thân phận của một số ô. Mục này trình bày thuật toán xác định số lượng nghiệm của một tập gợi ý hoàn chỉnh, được dùng để kiểm định bộ câu đố thử nghiệm.

\subsection{Phát biểu bài toán}
\label{subsec:phat-bieu-duy-nhat}

Khác với việc phân loại ở mục trước, việc kiểm tra tính duy nhất là kiểm tra \emph{ngoại tuyến}: đầu vào là \emph{tất cả} các gợi ý của câu đố mà không có thân phận nào được lật sẵn, và thuật toán không cần biết nghiệm ẩn của câu đố. Gọi $KB_{\text{đủ}}$ là CNF mã hóa toàn bộ các gợi ý. Kết quả thuộc một trong ba trường hợp:
\begin{itemize}
    \item \textsc{None}: $KB_{\text{đủ}}$ không thỏa được, các gợi ý mâu thuẫn nhau.
    \item \textsc{Unique}: $KB_{\text{đủ}}$ có đúng một mô hình, câu đố hợp lệ.
    \item \textsc{Multiple}: $KB_{\text{đủ}}$ có từ hai mô hình trở lên, câu đố mơ hồ.
\end{itemize}
Vì các biến phụ được xác định hàm theo các biến cơ sở, hai mô hình khác nhau luôn khác nhau ở ít nhất một biến cơ sở. Do đó có thể đếm nghiệm chỉ trên $n^2$ biến cơ sở.

\subsection{Thuật toán}
\label{subsec:thuat-toan-duy-nhat}

Cách đếm toàn bộ nghiệm có thể tốn số lần gọi bộ giải tỉ lệ với số nghiệm. Thuật toán dưới đây tránh điều đó bằng cách chỉ cần phân biệt ``một nghiệm'' với ``nhiều hơn một nghiệm''. Ý tưởng như sau: lấy một mô hình $M$ bất kỳ; $M$ là nghiệm duy nhất khi và chỉ khi mỗi ô bị \emph{ép} phải nhận đúng giá trị của nó trong $M$. Với mỗi ô ta hỏi liệu có mô hình nào gán cho ô đó giá trị ngược với $M$ hay không.

\begin{algorithm}[H]
\caption{Kiểm tra tính duy nhất \textsc{CheckUniqueness}}
\label{alg:check-uniqueness}
\begin{algorithmic}[1]
\REQUIRE $KB_{\text{đủ}}$ mã hóa toàn bộ gợi ý, đã nạp vào bộ giải; các biến cơ sở $v = 1, \dots, n^2$
\ENSURE \textsc{None}, \textsc{Unique} (kèm nghiệm) hoặc \textsc{Multiple} (kèm hai nghiệm khác nhau)
\STATE $(s, M) \leftarrow \textsc{Solve}([\,])$
\IF{$s$ không phải SAT}
    \RETURN \textsc{None}
\ENDIF
\FORALL{$v = 1, \dots, n^2$}
    \STATE $(s, M') \leftarrow \textsc{Solve}([\,\text{literal của } v \text{ ứng với giá trị } \neg M[v]\,])$
    \IF{$s$ là SAT}
        \RETURN \textsc{Multiple} với hai nghiệm $M$ và $M'$ \COMMENT{$M'[v] \ne M[v]$}
    \ENDIF
\ENDFOR
\RETURN \textsc{Unique} với nghiệm $M$
\end{algorithmic}
\end{algorithm}

\textbf{Tính đúng đắn.} Nếu thuật toán trả về \textsc{Multiple} thì $M$ và $M'$ đều là mô hình của $KB_{\text{đủ}}$ và khác nhau ở biến $v$, nên có ít nhất hai nghiệm; chính hai mô hình này là bằng chứng. Ngược lại, nếu có một nghiệm $M^* \ne M$ thì $M^*$ phải khác $M$ ở một biến cơ sở $v$ nào đó, khi đó truy vấn ứng với $v$ là SAT (mô hình $M^*$ làm chứng), nên thuật toán không thể trả về \textsc{Unique}. Vậy \textsc{Unique} chỉ được trả về khi $M$ là nghiệm duy nhất. Trường hợp \textsc{None} đúng theo định nghĩa của UNSAT.

\textbf{Số lần gọi bộ giải.} Tối đa $1 + n^2$ lần, tức là 10, 17 và 26 lần cho bàn $3\times3$, $4\times4$ và $5\times5$, không phụ thuộc vào số lượng nghiệm.

\section{Vòng lặp suy luận của tác tử logic}
\label{sec:vong-lap-suy-luan}

Tác tử suy luận logic (\texttt{LogicAgent}) là thành phần đóng vai trò trung tâm trong việc tự động giải quyết câu đố Griductive. Tác tử hoạt động hoàn toàn dựa trên cổng giao tiếp suy luận (\texttt{GamePort}) và bộ phân loại trạng thái logic (\texttt{EntailmentClassifier}), tuân thủ tuyệt đối các nguyên tắc thiết kế kiến trúc: không truy cập thông tin ẩn, suy luận tất định và không bao giờ phán đoán may rủi.

\subsection{Nguyên tắc suy luận không đoán mò (No-Guess Policy)}
\label{subsec:no-guess-policy}

Nguyên tắc cốt lõi chi phối mọi hành vi của tác tử là: \textbf{chỉ đưa ra phán quyết khi và chỉ khi thân phận của ô cờ là một hệ quả logic tất yếu ($KB_t \models \ell$)}. Cụ thể:
\begin{itemize}
    \item Nếu một ô cờ có trạng thái phân loại là \textsc{Unknown} (cả hai khả năng Tội phạm và Vô tội đều thỏa mãn $KB_t$), tác tử tuyệt đối không được phép "thử vận may" bằng cách chọn ngẫu nhiên một trong hai giá trị.
    \item Nếu tại một thời điểm $t$, toàn bộ các ô cờ chưa lật đều có trạng thái là \textsc{Unknown}, tác tử nhận diện rằng chuỗi suy luận đã rơi vào trạng thái bế tắc logic (\textsc{Stuck}) và lập tức dừng lại, trả về kết quả an toàn thay vì tiếp tục phân nhánh tìm kiếm mò mẫm.
    \item Cơ chế này bảo đảm rằng mọi phán quyết được gửi tới cổng trò chơi đều được chứng minh bằng toán học 100\%, ngăn ngừa tối đa nguy cơ bị phạt điểm hoặc nhận phán quyết sai do suy đoán vô căn cứ.
\end{itemize}

\subsection{Một bước suy luận đơn lẻ và trật tự chọn ô tất định}
\label{subsec:single-step-deduction}

Tại mỗi bước suy luận đơn lẻ, phương thức \texttt{step(port)} thực hiện một chu kỳ suy diễn khép kín gồm các giai đoạn sau:

\begin{enumerate}
    \item \textbf{Thu thập trạng thái công khai:} Lấy ảnh chụp trạng thái công khai $S_t = \text{port.get\_public\_state()}$. Nếu tập hợp các ô chưa lật $S_t.\text{unsolved}$ rỗng, trò chơi đã hoàn thành.
    \item \textbf{Phân loại trạng thái logic:} Gọi bộ phân loại suy diễn $\textsc{ClassifyState}(S_t)$ để thu được báo cáo phân loại trạng thái. Nếu $KB_t$ rơi vào trạng thái mâu thuẫn (\textsc{Inconsistent}) hoặc quá thời gian (\textsc{Timeout}), tác tử lập tức ghi nhận lỗi và dừng lại.
    \item \textbf{Lọc tập ô bị ép buộc (Forced cells):} Tập hợp toàn bộ các ô cờ chưa lật có thân phận đã được chứng minh bắt buộc:
    \[
    \mathcal{F} = \{c \in S_t.\text{unsolved} \mid \text{report.by\_cell}[c] \in \{\text{Criminal}, \text{Innocent}\}\}
    \]
    \item \textbf{Lựa chọn tất định theo thứ tự hàng trước (Row-Major Order):}
    Trong trường hợp có nhiều hơn một ô cờ cùng được chứng minh đồng thời ($|\mathcal{F}| > 1$), tác tử áp dụng quy tắc ưu tiên tất định: luôn chọn ô cờ có vị trí xuất hiện sớm nhất theo thứ tự quét từ trên xuống dưới, từ trái sang phải:
    \[
    c^* = \arg\min_{(r, c) \in \mathcal{F}} (r \cdot n + c)
    \]
    Quy tắc này bảo đảm tính nhất quán tuyệt đối: với cùng một trạng thái bàn cờ ban đầu, tác tử luôn thực hiện một chuỗi các bước đi duy nhất và có thể tái lập hoàn toàn trong các bài kiểm thử.
    \item \textbf{Gửi phán quyết nguyên tử kèm token:} Tác tử gửi phán quyết của ô $c^*$ tới cổng trò chơi kèm theo thẻ trạng thái mong đợi:
    \[
    \text{outcome} = \text{port.submit\_verdict}(c^*, v^*, \text{expected\_token} = S_t.\text{token})
    \]
    Việc đối soát \texttt{expected\_token} bảo đảm tính nguyên tử: nếu trạng thái bàn cờ đã bị thay đổi ngoài ý muốn bởi tiến trình khác, phán quyết sẽ bị từ chối bằng lỗi trạng thái lỗi thời (\texttt{StaleStateError}).
    \item \textbf{Ghi nhận vết suy luận (\textit{Deduction Trace}):} Toàn bộ thông tin của bước đi gồm số thứ tự bước, tọa độ ô $c^*$, phán quyết $v^*$, phản hồi chấp nhận từ động cơ, gợi ý mới được mở ra và thống kê truy vấn SAT được đóng gói vào đối tượng \texttt{StepResult} để phục vụ công tác gỡ lỗi, kiểm định và hiển thị trực quan.
\end{enumerate}

Thuật toán~\ref{alg:deductive-agent-step} mô tả chi tiết quy trình thực thi một bước của tác tử.

\begin{algorithm}[H]
\caption{Một bước suy luận của tác tử \textsc{DeductiveLogicAgentStep}}
\label{alg:deductive-agent-step}
\begin{algorithmic}[1]
\REQUIRE Cổng giao tiếp trò chơi $port$, bộ phân loại trạng thái $classifier$
\ENSURE Kết quả bước đi $step\_result$, hoặc trạng thái kết thúc (\textsc{Solved}, \textsc{Stuck}, \textsc{Inconsistent}, \textsc{Timeout})
\STATE $S \leftarrow port.\text{get\_public\_state}()$
\IF{$S.\text{unsolved} = \emptyset$}
    \RETURN Trạng thái \textsc{Solved}
\ENDIF
\STATE $report \leftarrow classifier.\text{classify\_state}(S)$
\IF{$report.\text{status} = \textsc{Inconsistent}$}
    \RETURN Trạng thái \textsc{Inconsistent}
\ELSIF{$report.\text{status} = \textsc{Timeout}$}
    \RETURN Trạng thái \textsc{Timeout}
\ENDIF
\STATE $\mathcal{F} \leftarrow [\,]$ \COMMENT{Danh sách các ô bị ép buộc theo thứ tự hàng trước}
\FORALL{$(r, c) \in S.\text{unsolved}$ theo thứ tự hàng trước}
    \IF{$report.\text{by\_cell}[(r, c)] \in \{\textsc{Criminal}, \textsc{Innocent}\}$}
        \STATE Thêm $((r, c), report.\text{by\_cell}[(r, c)])$ vào $\mathcal{F}$
    \ENDIF
\ENDFOR
\IF{$\mathcal{F}$ rỗng}
    \RETURN Trạng thái \textsc{Stuck} \COMMENT{Không có ô nào suy luận được, dừng lại an toàn}
\ENDIF
\STATE $((r^*, c^*), v^*) \leftarrow \mathcal{F}[0]$ \COMMENT{Chọn ô đầu tiên theo trật tự hàng trước}
\STATE $outcome \leftarrow port.\text{submit\_verdict}((r^*, c^*), v^*, S.\text{token})$
\STATE $step\_result \leftarrow \text{Khởi tạo } \texttt{StepResult}((r^*, c^*), v^*, outcome, report)$
\RETURN $step\_result$
\end{algorithmic}
\end{algorithm}

\subsection{Vòng lặp giải tự động toàn cục và vết suy luận}
\label{subsec:full-loop}

Vòng lặp tự động toàn cục của tác tử được hiện thực trong phương thức \texttt{run(port, max\_steps)}. Tại mỗi vòng lặp, tác tử gọi hàm \texttt{step(port)} và tích lũy vết suy luận cho đến khi gặp một trong các điều kiện dừng xác định trước.

\begin{algorithm}[H]
\caption{Vòng lặp giải toàn cục \textsc{DeductiveLogicAgentRun}}
\label{alg:deductive-agent-run}
\begin{algorithmic}[1]
\REQUIRE Cổng giao tiếp $port$, giới hạn số bước $max\_steps$
\ENSURE Bản ghi kết quả toàn cục \texttt{RunResult}
\STATE $trace \leftarrow [\,]$; $step\_count \leftarrow 0$
\WHILE{$step\_count < max\_steps$}
    \STATE $res \leftarrow \textsc{DeductiveLogicAgentStep}(port, classifier)$
    \IF{$res$ là kết quả hoàn thành hoặc lỗi dừng (\textsc{Solved}, \textsc{Stuck}, \textsc{Inconsistent}, \textsc{Timeout})}
        \STATE $final\_state \leftarrow port.\text{get\_public\_state}()$
        \RETURN \texttt{RunResult}($res$, $trace$, $final\_state$)
    \ENDIF
    \STATE Thêm $res$ vào danh sách $trace$
    \STATE $step\_count \leftarrow step\_count + 1$
\ENDWHILE
\RETURN \texttt{RunResult}(\textsc{Timeout}, $trace$, $port.\text{get\_public\_state}()$)
\end{algorithmic}
\end{algorithm}

Kết quả của toàn bộ quá trình chạy được chuẩn hóa trong đối tượng \texttt{RunResult}, lưu trữ:
\begin{itemize}
    \item Trạng thái dừng cuối cùng (\texttt{status}): Một trong các giá trị \textsc{Solved}, \textsc{Stuck}, \textsc{Inconsistent}, \textsc{Timeout}, hoặc \textsc{Rejected}.
    \item Toàn bộ chuỗi vết suy luận (\texttt{steps}): Danh sách tuần tự các đối tượng \texttt{StepResult}.
    \item Trạng thái bàn cờ cuối cùng (\texttt{final\_state}): Ảnh chụp bàn cờ khi dừng.
    \item Thống kê tổng hợp: Tổng số bước đi, tổng số lần gọi bộ giải SAT và tổng số phép lan truyền đơn vị đã tích lũy.
\end{itemize}

\subsection{Cơ chế gợi ý nước đi (Hint)}
\label{subsec:hint-mechanism}

Bên cạnh chế độ tự động giải toàn cục, tác tử cung cấp chức năng hỗ trợ người chơi thông qua cơ chế gợi ý nước đi (\texttt{hint}). Khi người chơi gặp khó khăn và yêu cầu trợ giúp, phương thức \texttt{hint(state)} tiến hành phân tích trạng thái công khai $S_t$ mà không thực hiện hành vi phán quyết hay làm thay đổi bàn cờ.

Quy trình gợi ý nước đi diễn ra như sau:
\begin{enumerate}
    \item Gọi bộ phân loại trạng thái $\textsc{ClassifyState}(S_t)$ trên ảnh chụp công khai hiện tại.
    \item Lọc và tìm ô cờ đầu tiên bị ép buộc theo thứ tự hàng trước $c^* \in \mathcal{F}$. Nếu không tìm thấy ô nào bị ép buộc, cơ chế gợi ý thông báo không có bước đi tất định khả dụng.
    \item \textbf{Truy vết các manh mối liên quan:} Sử dụng hàm phân tích hình học \texttt{referenced\_cells} (trong mô-đun \texttt{common.regions}) để xác định tập hợp các ô cờ công khai mang gợi ý có vùng tham chiếu giao cắt với vị trí ô $c^*$. Đây chính là các dữ kiện tiên đề cấu thành nên chứng minh logic cho ô $c^*$.
    \item Đóng gói và trả về đối tượng \texttt{Hint(token, cell, verdict, clue\_ids)}, bao gồm tọa độ ô cần đi tiếp, danh tính bắt buộc và danh sách các mã gợi ý làm bằng chứng.
\end{enumerate}

Chức năng gợi ý đóng vai trò là một trợ thủ suy luận thuần túy (\textit{Pure Query}), bảo đảm tính toàn vẹn của trạng thái trò chơi và cung cấp căn cứ minh bạch giúp người chơi hiểu rõ lý do đằng sau mỗi bước suy luận.
